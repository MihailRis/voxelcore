# Библиотека *time*

```lua
-- Возвращает время с момента запуска движка в секундах и миллисекундах.
time.uptime() -> number

-- Возвращает дельту времени в секундах и миллисекундах (время прошедшее с предыдущего кадра)
time.delta() -> number

-- Возвращает время UTC (секунды с 1970-01-01 UTC)
time.utc_time() -> int

-- Возвращает время UTC (секунды с 1970-01-01 UTC) с точностью до миллисекунд
time.precise_utc_time() -> number

-- Возвращает локальное (системное) время в секундах
time.local_time() -> int

-- Возвращает смещение локального времени от UTC в секундах
time.utc_offset() -> int

-- Возвращает время высокой точности в секундах. За точку отсчёта берётся запуск движка.
time.precise_time() -> number
```
